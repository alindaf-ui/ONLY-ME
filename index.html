// ---------- GET TOP SELLERS (laaste 30 dae) ---------- function getTopSellers(ss, params) { var sheet = ss.getSheetByName('SALES LOG'); if (!sheet) return { status: 'ok', data: [] }; var lastRow = sheet.getLastRow(); if (lastRow < 4) return { status: 'ok', data: [] }; var days = parseInt((params && params.days) || 30, 10); var cutoff = new Date(); cutoff.setDate(cutoff.getDate() - days); var range = sheet.getRange(4, 1, lastRow - 3, 13); var values = range.getValues(); var totals = {}; for (var i = 0; i < values.length; i++) { var row = values[i]; var dateVal = row[0]; var sku = String(row[2] || '').trim(); var supplier = String(row[3] || '').trim(); var category = String(row[4] || '').trim(); var product = String(row[5] || '').trim(); var qty = parseFloat(row[6]) || 0; var lineTotal = parseFloat(row[10]) || 0; if (!product || qty <= 0) continue; var d = dateVal instanceof Date ? dateVal : new Date(dateVal); if (isNaN(d.getTime()) || d < cutoff) continue; var key = sku || product; if (!totals[key]) { totals[key] = { sku: sku, supplier: supplier, category: category, product: product, qty: 0, revenue: 0, timesSold: 0 }; } totals[key].qty += qty; totals[key].revenue += lineTotal; totals[key].timesSold += 1; } var list = Object.keys(totals).map(function(k) { return totals[k]; }); list.sort(function(a, b) { return b.qty - a.qty; }); return { status: 'ok', data: list, days: days }; }
